cmake_minimum_required(VERSION 3.16)
project(vns LANGUAGES CXX C)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Compiler warnings
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -Wpedantic")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -Wall -Wextra -Wpedantic")

# Build types
# -DDEBUG is intentionally NOT defined: it collides with LogLevel::DEBUG.
set(CMAKE_CXX_FLAGS_DEBUG "-g -O0")
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -DNDEBUG")

# Project version
set(VNS_VERSION_MAJOR 1)
set(VNS_VERSION_MINOR 0)
set(VNS_VERSION_PATCH 0)

# Include directories
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)

# Find required packages
find_package(Threads REQUIRED)

# Core library (shared code)
add_library(vns_core STATIC
    src/network/ip_address.cpp
    src/network/cidr.cpp
    src/network/network.cpp
    src/network/device.cpp
    src/nat/nat_engine.cpp
    src/nat/nat.cpp
    src/nat/nat_table.cpp
    src/nat/pat_allocator.cpp
    src/nat/port_forwarding.cpp
    src/nat/dnat.cpp
    src/packet/packet.cpp
    src/packet/packet_engine.cpp
    src/connection/connection_tracker.cpp
    src/driver/driver_interface.cpp
    src/utils/logger.cpp
    src/services/simulation_service.cpp
)

target_link_libraries(vns_core PRIVATE Threads::Threads)

# Main executable
add_executable(vns_sim
    src/main.cpp
    src/cli/cli.cpp
    src/cli/formatter.cpp
)

target_link_libraries(vns_sim PRIVATE vns_core Threads::Threads)
target_compile_definitions(vns_sim PRIVATE
    VNS_VERSION_STRING="${VNS_VERSION_MAJOR}.${VNS_VERSION_MINOR}.${VNS_VERSION_PATCH}"
)

# Tests - each test file provides its own main(), so build one executable per file.
enable_testing()
set(VNS_TEST_NAMES
    network
    nat
    pat
    dnat
    packet
    connection
    driver_interface
)

set(VNS_TEST_TARGETS "")
foreach(test_name ${VNS_TEST_NAMES})
    set(test_target "test_${test_name}")
    add_executable(${test_target} tests/test_${test_name}.cpp)
    target_link_libraries(${test_target} PRIVATE vns_core Threads::Threads)
    # The suites are written with assert(); keep them active even in Release
    # (-DNDEBUG) so the tests cannot silently pass by checking nothing.
    target_compile_options(${test_target} PRIVATE -UNDEBUG)
    add_test(NAME ${test_target} COMMAND ${test_target})
    list(APPEND VNS_TEST_TARGETS ${test_target})
endforeach()

# Convenience runner: `./vns_tests` executes every suite in order.
set(_runner "#!/usr/bin/env bash\n# Auto-generated by CMake - runs the full VNS test suite.\nset -e\nDIR=\"\$(cd \"\$(dirname \"\$0\")\" && pwd)\"\n")
foreach(test_target ${VNS_TEST_TARGETS})
    string(APPEND _runner "echo \"==> ${test_target}\"\n\"\$DIR/${test_target}\"\n")
endforeach()
string(APPEND _runner "echo \"All test suites passed.\"\n")
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/vns_tests "${_runner}")
execute_process(COMMAND chmod +x ${CMAKE_CURRENT_BINARY_DIR}/vns_tests)

# Install
install(TARGETS vns_sim DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)

# Custom targets
add_custom_target(build-driver
    COMMAND make -C ${CMAKE_CURRENT_SOURCE_DIR}/kernel/vns_control
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/kernel/vns_control
    COMMENT "Building kernel module vns_control"
)

add_custom_target(load-driver
    COMMAND sudo insmod ${CMAKE_CURRENT_SOURCE_DIR}/kernel/vns_control/vns_control.ko
    COMMENT "Loading vns_control kernel module"
    DEPENDS build-driver
)

add_custom_target(unload-driver
    COMMAND sudo rmmod vns_control
    COMMENT "Unloading vns_control kernel module"
)

add_custom_target(driver-stats
    COMMAND ${CMAKE_CURRENT_BINARY_DIR}/vns_sim --driver-stats
    COMMENT "Show driver statistics"
    DEPENDS vns_sim
)

# Clean target enhancement
add_custom_target(clean-all
    COMMAND make -C ${CMAKE_CURRENT_SOURCE_DIR}/kernel/vns_control clean
    COMMENT "Cleaning kernel module build"
)
